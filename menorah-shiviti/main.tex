% menorah-shiviti/main.tex
% Menorah Shiviti: Psalm 67 in the shape of a menorah. DRAFT sketch.
% Compile with LuaLaTeX (twice: the drawing is positioned on the page).

% Output format: aceo, siddur, birkon or poster (sizes in menorah.lua).
\providecommand{\menorahformat}{aceo}

\documentclass{article}

% Locate the repository root so the shared files resolve whether this file
% is compiled from menorah-shiviti/ (locally) or from the project root
% (Overleaf).
\IfFileExists{../shared/macros/hebrew-layout.sty}
  {\def\pressroot{../}}
  {\def\pressroot{}}
\makeatletter
\def\input@path{{\pressroot shared/macros/}}
\makeatother

\usepackage{geometry}
\usepackage{hebrew-layout}
\usepackage{graphicx}
\usepackage{tikz}

% Hebrew face for this piece: Stam Sefarad CLM (Culmus), a Sephardic
% scribal (STaM) hand. It ships with TeX Live, so it is not copied into the
% repo: a second copy under the same name confuses the LuaTeX font cache.
\newfontfamily\stamfont{StamSefaradCLM.ttf}[Script=Hebrew, Renderer=HarfBuzz]

\directlua{
  menorah = dofile("\pressroot menorah-shiviti/menorah.lua")
  menorah.setup("\menorahformat", "\pressroot menorah-shiviti/")
}

\begin{document}
\pagestyle{empty}
\directlua{menorah.measure()}
\noindent\directlua{menorah.draw()}
\end{document}
