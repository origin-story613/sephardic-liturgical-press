% modim-derabanan/main.tex
% Modim deRabanan poster: a formal text panel. DRAFT.
% Compile with LuaLaTeX (twice: the frame is placed relative to the page).

% Output format: aceo, siddur, birkon or poster.
\providecommand{\modimformat}{poster}

\documentclass{article}

% Locate the repository root so the shared files resolve whether this file
% is compiled from modim-derabanan/ (locally) or from the project root
% (Overleaf).
\IfFileExists{../shared/macros/hebrew-layout.sty}
  {\def\pressroot{../}}
  {\def\pressroot{}}
\makeatletter
\def\input@path{{\pressroot shared/macros/}}
\makeatother

\usepackage{geometry}
\usepackage{hebrew-layout}   % Hebrew in Taamey David CLM, with nikud
\usepackage{tikz}

% Everything scales with one unit: 1% of the width of the largest box of
% the poster's proportions (2:3) that fits the page, so every format has the
% same design whatever its page shape.
\newcommand{\textsizefactor}{6.0}             % body text, in units
\newcommand{\headingfactor}{8.5}              % heading, in units
\newcommand{\framemargin}{6}                  % page edge to frame, in units
\newcommand{\textmargin}{6}                   % frame to text, in units

% Page sizes in inches: the same formats as the shivitis. The text margin
% is the frame margin plus the gap inside the frame.
\directlua{
  local f = ({ aceo = { 2.5, 3.5 }, siddur = { 3.5, 4.75 },
               birkon = { 4.25, 5.5 }, poster = { 12, 18 } })["\modimformat"]
  local u = math.min(f[1], f[2] / 1.5) / 100
  local m = u * (\framemargin + \textmargin)
  tex.sprint("\string\\geometry{paperwidth=" .. f[1] .. "in,paperheight="
    .. f[2] .. "in,margin=" .. m .. "in}")
  tex.sprint("\string\\newlength{\string\\unit}\string\\setlength{\string\\unit}{" .. u .. "in}")
}

\begin{document}
\pagestyle{empty}
% Only scaled spacing on this page: the house style's paragraph gap and the
% centre environment's space are fixed sizes, too big on an ACEO card.
\setlength{\parskip}{0pt}

\vspace*{\fill}
{\centering
  {\fontsize{\headingfactor\unit}{1.3\dimexpr\headingfactor\unit\relax}\selectfont\bfseries
   % the frame, drawn from inside the heading line so it takes no space
   \begin{tikzpicture}[remember picture,overlay]
     \draw[line width=0.12\unit]
       ([xshift=\framemargin\unit,yshift=\framemargin\unit]current page.south west)
       rectangle
       ([xshift=-\framemargin\unit,yshift=-\framemargin\unit]current page.north east);
   \end{tikzpicture}%
   מוֹדִים דְּרַבָּנָן\par}
  \vspace{3\unit}
  {\fontsize{\textsizefactor\unit}{1.65\dimexpr\textsizefactor\unit\relax}\selectfont
   \input{\pressroot modim-derabanan/text}\par}
}
\vfill

\end{document}
