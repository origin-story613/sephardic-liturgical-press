\sectiontitle{אַשְׁרֵי}

\textbf{אַשְׁרֵי} יוֹשְׁבֵי בֵיתֶךָ עוֹד יְהַלְלוּךָ סֶּלָה: אַשְׁרֵי הָעָם שֶׁכָּכָה לּוֹ אַשְׁרֵי הָעָם שֶׁ\Shem{} אֱלֹהָיו:

\textbf{תְּהִלָּה} לְדָוִד \textbf{אֲ}רוֹמִמְךָ אֱלוֹהַי הַמֶּלֶךְ וַאֲבָרְכָה שִׁמְךָ לְעוֹלָם וָעֶד: \textbf{בְּ}כָל יוֹם אֲבָרְכֶךָּ וַאֲהַלְלָה שִׁמְךָ לְעוֹלָם וָעֶד: \textbf{גָּ}דוֹל \Shem{} וּמְהֻלָּל מְאֹד וְלִגְדֻלָּתוֹ אֵין חֵקֶר: \textbf{דּ}וֹר לְדוֹר יְשַׁבַּח מַעֲשֶׂיךָ וּגְבוּרֹתֶיךָ יַגִּידוּ: \textbf{הֲ}דַר כְּבוֹד הוֹדֶךָ וְדִבְרֵי נִפְלְאוֹתֶיךָ אָשִׂיחָה: \textbf{וֶ}עֱזוּז נוֹרְאֹתֶיךָ יֹאמֵרוּ וּגְדוּלָּתְךָ אֲסַפְּרֶנָּה: \textbf{זֵ}כֶר רַב טוּבְךָ יַבִּיעוּ וְצִדְקָתְךָ יְרַנֵּנוּ: \textbf{חַ}נּוּן וְרַחוּם \Shem{} אֶרֶךְ אַפַּיִם וּגְדָל חָסֶד: \textbf{ט}וֹב \Shem{} לַכֹּל וְרַחֲמָיו עַל כָּל מַעֲשָׂיו: \textbf{י}וֹדוּךָ \Shem{} כָּל מַעֲשֶׂיךָ וַחֲסִידֶיךָ יְבָרְכוּכָה: \textbf{כְּ}בוֹד מַלְכוּתְךָ יֹאמֵרוּ וּגְבוּרָתְךָ יְדַבֵּרוּ: \textbf{לְ}הוֹדִיעַ לִבְנֵי הָאָדָם גְּבוּרֹתָיו וּכְבוֹד הֲדַר מַלְכוּתוֹ: \textbf{מַ}לְכוּתְךָ מַלְכוּת כָּל עֹלָמִים וּמֶמְשֶׁלְתְּךָ בְּכָל דּוֹר וָדוֹר: \textbf{ס}וֹמֵךְ \Shem{} לְכָל הַנֹּפְלִים וְזוֹקֵף לְכָל הַכְּפוּפִים: \textbf{עֵ}ינֵי כֹל אֵלֶיךָ יְשַׂבֵּרוּ וְאַתָּה נוֹתֵן לָהֶם אֶת אָכְלָם בְּעִתּוֹ: \textbf{פּ}וֹתֵחַ אֶת יָדֶךָ וּמַשְׂבִּיעַ לְכָל חַי רָצוֹן: \textbf{צַ}דִּיק \Shem{} בְּכָל דְּרָכָיו וְחָסִיד בְּכָל מַעֲשָׂיו: \textbf{קָ}רוֹב \Shem{} לְכָל קֹרְאָיו לְכָל אֲשֶׁר יִקְרָאֻהוּ בֶאֱמֶת: \textbf{רְ}צוֹן יְרֵאָיו יַעֲשֶׂה וְאֶת שַׁוְעָתָם יִשְׁמַע וְיוֹשִׁיעֵם: \textbf{שׁ}וֹמֵר \Shem{} אֶת כָּל אֹהֲבָיו וְאֵת כָּל הָרְשָׁעִים יַשְׁמִיד: \textbf{תְּ}הִלַּת \Shem{} יְדַבֶּר פִּי וִיבָרֵךְ כָּל בָּשָׂר שֵׁם קָדְשׁוֹ לְעוֹלָם וָעֶד: וַאֲנַחְנוּ נְבָרֵךְ יָהּ מֵעַתָּה וְעַד עוֹלָם הַלְלוּיָהּ:

\divider

\sectiontitle{וּבָא לְצִיּוֹן}

\rubric{The Aramaic passages in parentheses are said quietly.}
\textbf{וּבָא לְצִיּוֹן} גּוֹאֵל וּלְשָׁבֵי פֶשַׁע בְּיַעֲקֹב נְאֻם \Shem{}: וַאֲנִי זֹאת בְּרִיתִי אוֹתָם אָמַר \Shem{} רוּחִי אֲשֶׁר עָלֶיךָ וּדְבָרַי אֲשֶׁר שַׂמְתִּי בְּפִיךָ לֹא יָמוּשׁוּ מִפִּיךָ וּמִפִּי זַרְעֲךָ וּמִפִּי זֶרַע זַרְעֲךָ אָמַר \Shem{} מֵעַתָּה וְעַד עוֹלָם: וְאַתָּה קָדוֹשׁ יוֹשֵׁב תְּהִלּוֹת יִשְׂרָאֵל: וְקָרָא זֶה אֶל זֶה וְאָמַר קָדוֹשׁ קָדוֹשׁ קָדוֹשׁ \Shem{} צְבָאוֹת מְלֹא כָל הָאָרֶץ כְּבוֹדוֹ: \whispered{וּמְקַבְּלִין דֵין מִן דֵין וְאָמְרִין קָדִיש בִּשְמֵי מֵרוֹמַא עִלָאַה בֵּית שְכִינְתֵּהּ, קַדִּישׁ עַל אַרְעָא עובַד גְּבוּרְתֵּהּ, קַדִּישׁ לְעָלַם וּלְעָלְמֵי עָלְמַיָּא, \Shem{} צְבָאוֹת מַלְיָא כָּל אַרְעָא זִיו יִקְרֵיהּ:} וַתִּשָּׂאֵנִי רוּחַ וָאֶשְׁמַע אַחֲרַי קוֹל רַעַשׁ גָּדוֹל בָּרוּךְ כְּבוֹד \Shem{} מִמְּקוֹמוֹ: \whispered{וּנְטָלַתְנִי רוּחָא וּשְׁמָעִית בַּתְרַי קַל זִיעַ שַׂגִּיא דִמְשַׁבְּחִין וְאָמְרִין, בְּרִיךְ יְקָרָא דְ\Shem{} מֵאֲתָר בֵּית שְכִינְתֵּהּ, \Shem{} יִמְלוֹךְ לְעוֹלָם וָעֶד, \Shem{} מַלְכוּתֵיהּ קָאִים לְעָלַם וּלְעָלְמֵי עָלְמַיָּא:} \Shem{} אֱלֹהֵי אַבְרָהָם יִצְחָק וְיִשְׂרָאֵל אֲבוֹתֵינוּ, שָׁמְרָה זֹאת לְעוֹלָם לְיֵצֶר מַחְשְׁבוֹת לְבַב עַמֶּךָ, וְהָכֵן לְבָבָם אֵלֶיךָ: וְהוּא רַחוּם יְכַפֵּר עָווֹן וְלֹא יַשְׁחִית, וְהִרְבָּה לְהָשִׁיב אַפּוֹ, וְלֹא יָעִיר כָּל חֲמָתוֹ: כִּי אַתָּה \Shem{} טוֹב וְסַלָּח וְרַב חֶסֶד לְכָל קוֹרְאֶיךָ: צִדְקָתְךָ צֶּדֶק לְעוֹלָם, וְתוֹרָתְךָ אֱמֶת: תִתֵּן אֱמֶת לְיַעֲקֹב, חֶסֶד לְאַבְרָהָם, אֲשֶׁר נִשְׁבַּעְתָּ לַאֲבֹתֵינוּ מִימֵי קֶדֶם: בָּרוּךְ אֲדֹנָי יוֹם יוֹם, יַעֲמָס לָנוּ הָאֵל יְשׁוּעָתֵנוּ סֶלָה: \Shem{} צְבָאוֹת עִמָּנוּ מִשְׂגָּב לָנוּ אֱלֹהֵי יַעֲקֹב סֶלָה: \Shem{} צְבָאוֹת אַשְׁרֵי אָדָם בֹּטֵחַ בָּךְ: \Shem{} הוֹשִׁיעָה הַמֶּלֶךְ יַעֲנֵנוּ בְיוֹם קָרְאֵנוּ:

\textbf{בָּרוּךְ אֱלֹהֵינוּ} שֶׁבְּרָאָנוּ לִכְבוֹדוֹ, וְהִבְדִּילָנוּ מִן הַתּוֹעִים, וְנָתַן לָנוּ תּוֹרַת אֱמֶת, וְחַיֵּי עוֹלָם נָטַע בְּתוֹכֵנוּ. הוּא יִפְתַּח לִבֵּנוּ בְּתוֹרָתוֹ, וְיָשִׂים בְּלִבֵּנוּ אַהֲבָתוֹ וְיִרְאָתוֹ לַעֲשׂוֹת רְצוֹנוֹ וּלְעָבְדוֹ בְּלֵבָב שָׁלֵם. לֹא נִיגַע לָרִיק, וְלֹא נֵלֵד לַבֶּהָלָה. יְהִי רָצוֹן מִלְּפָנֶיךָ \Shem{} אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ שֶׁנִּשְׁמֹר חֻקֶּיךָ וּמִצְוֹתֶיךָ בָּעוֹלָם הַזֶּה, וְנִזְכֶּה וְנִחְיֶה וְנִירַשׁ טוֹבָה וּבְרָכָה לְחַיֵּי הָעוֹלָם הַבָּא: לְמַעַן יְזַמֶּרְךָ כָבוֹד וְלֹא יִדֹּם, \Shem{} אֱלֹהַי לְעוֹלָם אוֹדֶךָּ: \Shem{} חָפֵץ לְמַעַן צִדְקוֹ, יַגְדִּיל תּוֹרָה וְיַאְדִּיר: וְיִבְטְחוּ בְךָ יוֹדְעֵי שְׁמֶךָ, כִּי לֹא עָזַבְתָּ דֹרְשֶׁיךָ \Shem{}: \Shem{} אֲדֹנֵינוּ, מָה אַדִּיר שִׁמְךָ בּכָל הָאָרֶץ: חִזְקוּ וְיַאֲמֵץ לְבַבְכֶם כָּל הַמְיַחֲלִים \Shem{לַ}:
