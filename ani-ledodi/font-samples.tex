% One card: letter and name, the 2x2 block (right to left: א ה / ב ה),
% and the sample text at two sizes.
\newcommand{\card}[3]{%
  \begin{minipage}[t][2.45in][t]{0.48\textwidth}\centering
    {\foreignlanguage{english}{\textbf{#1)} #3}\par}
    \vspace{4pt}
    {#2\color{ahavagold}\fontsize{44pt}{46pt}\selectfont אה\\בה\par}
    \vspace{4pt}
    {#2\fontsize{8pt}{10pt}\selectfont\sample\par}
    \vspace{2pt}
    {#2\fontsize{5pt}{6.3pt}\selectfont\sample\par}
  \end{minipage}}
