\sectiontitle{קִדּוּשׁ יוֹם שַׁבָּת}\label{kiddush-shabbat-day}

\textbf{מִזְמוֹר לְדָוִד}, \Shem{} רֹעִי לֹא אֶחְסָר: בִּנְאוֹת דֶּשֶׁא יַרְבִּיצֵנִי, עַל מֵי מְנֻחוֹת יְנַהֲלֵנִי: נַפְשִׁי יְשׁוֹבֵב, יַנְחֵנִי בְּמַעְגְּלֵי צֶדֶק לְמַעַן שְׁמוֹ: גַּם כִּי אֵלֵךְ בְּגֵיא צַלְמָוֶת לֹא אִירָא רָע כִּי אַתָּה עִמָּדִי, שִׁבְטְךָ וּמִשְׁעַנְתֶּךָ הֵמָּה יְנַחֲמֻנִי: תַּעֲרֹךְ לְפָנַי שֻׁלְחָן נֶגֶד צֹרְרָי, דִּשַּׁנְתָּ בַשֶּׁמֶן רֹאשִׁי כּוֹסִי רְוָיָה: אַךְ טוֹב וָחֶסֶד יִרְדְּפוּנִי כָּל יְמֵי חַיָּי, וְשַׁבְתִּי בְּבֵית \Shem{} לְאֹרֶךְ יָמִים:

\textbf{אִם תָּשִׁיב} מִשַּׁבָּת רַגְלֶךָ, עֲשׂוֹת חֲפָצֶךָ בְּיוֹם קָדְשִׁי, וְקָרָאתָ לַשַּׁבָּת עֹנֶג לִקְדוֹשׁ \Shem{} מְכֻבָּד, וְכִבַּדְתּוֹ מֵעֲשׂוֹת דְּרָכֶיךָ, מִמְּצוֹא חֶפְצְךָ וְדַבֵּר דָּבָר: אָז תִּתְעַנַּג עַל \Shem{} וְהִרְכַּבְתִּיךָ עַל בָּמֳתֵי אָרֶץ, וְהַאֲכַלְתִּיךָ נַחֲלַת יַעֲקֹב אָבִיךָ כִּי פִּי \Shem{} דִּבֵּר:

\textbf{וְשָׁמְרוּ} בְנֵי יִשְׂרָאֵל אֶת הַשַּׁבָּת, לַעֲשׂוֹת אֶת הַשַּׁבָּת לְדֹרֹתָם בְּרִית עוֹלָם: בֵּינִי וּבֵין בְּנֵי יִשְׂרָאֵל אוֹת הִיא לְעֹלָם, כִּי שֵׁשֶׁת יָמִים עָשָׂה \Shem{} אֶת הַשָּׁמַיִם וְאֶת הָאָרֶץ וּבַיּוֹם הַשְּׁבִיעִי שָׁבַת וַיִּנָּפַשׁ: עַל כֵּן בֵּרַךְ \Shem{} אֶת יוֹם הַשַּׁבָּת וַיְקַדְּשֵׁהוּ:

\rubric{The one making Kiddush says, even when alone:}
סַבְרֵי מָרָנָן.

\rubric{The others respond:}
לְחַיִּים.

\textbf{בָּרוּךְ} אַתָּה \Shem{}, אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, בּוֹרֵא פְּרִי הַגֶּפֶן:

\rubric{Drink a revi'it of wine, or at least a cheekful, and give some to everyone at the table. Then wash and say Hamotzi (page~\pageref{netilat-yadayim}).}
