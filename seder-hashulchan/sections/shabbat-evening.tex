\sectiontitle{שָׁלוֹם עֲלֵיכֶם}

\rubric{Each verse is said three times.}
\textbf{שָׁלוֹם עֲלֵיכֶם} מַלְאֲכֵי הַשָּׁרֵת, מַלְאֲכֵי עֶלְיוֹן, מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא:
\textbf{שָׁלוֹם עֲלֵיכֶם} מַלְאֲכֵי הַשָּׁרֵת, מַלְאֲכֵי עֶלְיוֹן, מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא:
\textbf{שָׁלוֹם עֲלֵיכֶם} מַלְאֲכֵי הַשָּׁרֵת, מַלְאֲכֵי עֶלְיוֹן, מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא:

\textbf{בּוֹאֲכֶם לְשָׁלוֹם} מַלְאֲכֵי הַשָּׁלוֹם, מַלְאֲכֵי עֶלְיוֹן, מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא:
\textbf{בּוֹאֲכֶם לְשָׁלוֹם} מַלְאֲכֵי הַשָּׁלוֹם, מַלְאֲכֵי עֶלְיוֹן, מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא:
\textbf{בּוֹאֲכֶם לְשָׁלוֹם} מַלְאֲכֵי הַשָּׁלוֹם, מַלְאֲכֵי עֶלְיוֹן, מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא:

\textbf{בָּרְכוּנוּ לְשָׁלוֹם} מַלְאֲכֵי הַשָּׁלוֹם, מַלְאֲכֵי עֶלְיוֹן, מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא:
\textbf{בָּרְכוּנוּ לְשָׁלוֹם} מַלְאֲכֵי הַשָּׁלוֹם, מַלְאֲכֵי עֶלְיוֹן, מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא:
\textbf{בָּרְכוּנוּ לְשָׁלוֹם} מַלְאֲכֵי הַשָּׁלוֹם, מַלְאֲכֵי עֶלְיוֹן, מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא:

\textbf{בְּשִׁבְתְּכֶם לְשָׁלוֹם} מַלְאֲכֵי הַשָּׁלוֹם, מַלְאֲכֵי עֶלְיוֹן, מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא:
\textbf{בְּשִׁבְתְּכֶם לְשָׁלוֹם} מַלְאֲכֵי הַשָּׁלוֹם, מַלְאֲכֵי עֶלְיוֹן, מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא:
\textbf{בְּשִׁבְתְּכֶם לְשָׁלוֹם} מַלְאֲכֵי הַשָּׁלוֹם, מַלְאֲכֵי עֶלְיוֹן, מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא:

\textbf{בְּצֵאתְכֶם לְשָׁלוֹם} מַלְאֲכֵי הַשָּׁלוֹם, מַלְאֲכֵי עֶלְיוֹן, מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא:
\textbf{בְּצֵאתְכֶם לְשָׁלוֹם} מַלְאֲכֵי הַשָּׁלוֹם, מַלְאֲכֵי עֶלְיוֹן, מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא:
\textbf{בְּצֵאתְכֶם לְשָׁלוֹם} מַלְאֲכֵי הַשָּׁלוֹם, מַלְאֲכֵי עֶלְיוֹן, מֶלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא:

\textbf{כִּי} מַלְאָכָיו יְצַוֶּה לָּךְ, לִשְׁמָרְךָ בְּכָל דְּרָכֶיךָ: \Shem{} יִשְׁמֹר צֵאתְךָ וּבוֹאֶךָ, מֵעַתָּה וְעַד עוֹלָם:

\sectiontitle{אֵשֶׁת חַיִל}

\textbf{אֵשֶׁת חַיִל} מִי יִמְצָא וְרָחֹק מִפְּנִינִים מִכְרָהּ: \textbf{בָּ}טַח בָּהּ לֵב בַּעְלָהּ וְשָׁלָל לֹא יֶחְסָר: \textbf{גְּ}מָלַתְהוּ טוֹב וְלֹא רָע כָּל יְמֵי חַיֶּיהָ: \textbf{דָּ}רְשָׁה צֶמֶר וּפִשְׁתִּים וַתַּעַשׂ בְּחֵפֶץ כַּפֶּיהָ: \textbf{הָ}יְתָה כָּאֳנִיּוֹת סוֹחֵר מִמֶּרְחָק תָּבִיא לַחְמָהּ: \textbf{וַ}תָּקָם בְּעוֹד לַיְלָה וַתִּתֵּן טֶרֶף לְבֵיתָהּ וְחֹק לְנַעֲרֹתֶיהָ: \textbf{זָ}מְמָה שָׂדֶה וַתִּקָּחֵהוּ מִפְּרִי כַפֶּיהָ נָטְעָה כָּרֶם: \textbf{חָ}גְרָה בְעוֹז מָתְנֶיהָ וַתְּאַמֵּץ זְרֹעוֹתֶיהָ: \textbf{טָ}עֲמָה כִּי טוֹב סַחְרָהּ לֹא יִכְבֶּה בַלַּיְלָה נֵרָהּ: \textbf{יָ}דֶיהָ שִׁלְּחָה בַכִּישׁוֹר וְכַפֶּיהָ תָּמְכוּ פָלֶךְ: \textbf{כַּ}פָּהּ פָּרְשָׂה לֶעָנִי וְיָדֶיהָ שִׁלְּחָה לָאֶבְיוֹן: \textbf{לֹ}א תִירָא לְבֵיתָהּ מִשָּׁלֶג כִּי כָל בֵּיתָהּ לָבֻשׁ שָׁנִים: \textbf{מַ}רְבַדִּים עָשְׂתָה לָּהּ שֵׁשׁ וְאַרְגָּמָן לְבוּשָׁהּ: \textbf{נ}וֹדָע בַּשְּׁעָרִים בַּעְלָהּ בְּשִׁבְתּוֹ עִם זִקְנֵי אָרֶץ: \textbf{סָ}דִין עָשְׂתָה וַתִּמְכֹּר וַחֲגוֹר נָתְנָה לַכְּנַעֲנִי: \textbf{עֹ}ז וְהָדָר לְבוּשָׁהּ וַתִּשְׂחַק לְיוֹם אַחֲרוֹן: \textbf{פִּ}יהָ פָּתְחָה בְחָכְמָה וְתוֹרַת חֶסֶד עַל לְשׁוֹנָהּ: \textbf{צ}וֹפִיָּה הֲלִיכוֹת בֵּיתָהּ וְלֶחֶם עַצְלוּת לֹא תֹאכֵל: \textbf{קָ}מוּ בָנֶיהָ וַיְאַשְּׁרוּהָ בַּעְלָהּ וַיְהַלְלָהּ: \textbf{רַ}בּוֹת בָּנוֹת עָשׂוּ חָיִל וְאַתְּ עָלִית עַל כֻּלָּנָה: \textbf{שֶׁ}קֶר הַחֵן וְהֶבֶל הַיֹּפִי אִשָּׁה יִרְאַת \Shem{} הִיא תִתְהַלָּל: \textbf{תְּ}נוּ לָהּ מִפְּרִי יָדֶיהָ וִיהַלְלוּהָ בַשְּׁעָרִים מַעֲשֶׂיהָ:

\sectiontitle{קִדּוּשׁ לֵיל שַׁבָּת}\label{kiddush-shabbat-night}

\textbf{י}וֹם \textbf{הַ}שִּׁשִּׁי: \textbf{וַ}יְכֻלּוּ \textbf{הַ}שָּׁמַיִם וְהָאָרֶץ וְכָל צְבָאָם: וַיְכַל אֱלֹהִים בַּיּוֹם הַשְּׁבִיעִי מְלַאכְתּוֹ אֲשֶׁר עָשָׂה וַיִּשְׁבֹּת בַּיּוֹם הַשְּׁבִיעִי מִכָּל מְלַאכְתּוֹ אֲשֶׁר עָשָׂה: וַיְבָרֶךְ אֱלֹהִים אֶת יוֹם הַשְּׁבִיעִי וַיְקַדֵּשׁ אֹתוֹ כִּי בוֹ שָׁבַת מִכָּל מְלַאכְתּוֹ אֲשֶׁר בָּרָא אֱלֹהִים לַעֲשׂוֹת:

\rubric{The one making Kiddush says:}
סַבְרֵי מָרָנָן.

\rubric{The others respond:}
לְחַיִּים.

\textbf{בָּרוּךְ} אַתָּה \Shem{}, אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, בּוֹרֵא פְּרִי הַגֶּפֶן:

\textbf{בָּרוּךְ} אַתָּה \Shem{}, אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, אֲשֶׁר קִדְּשָׁנוּ בְּמִצְוֹתָיו, וְרָצָה בָנוּ, וְשַׁבַּת קָדְשׁוֹ בְּאַהֲבָה וּבְרָצוֹן הִנְחִילָנוּ, זִכָּרוֹן לְמַעֲשֵׂה בְרֵאשִׁית, תְּחִלָּה לְמִקְרָאֵי קֹדֶשׁ, זֵכֶר לִיצִיאַת מִצְרַיִם. וְשַׁבַּת קָדְשְׁךָ בְּאַהֲבָה וּבְרָצוֹן הִנְחַלְתָּנוּ. בָּרוּךְ אַתָּה \Shem{}, מְקַדֵּשׁ הַשַּׁבָּת:

\rubric{Sit and drink a revi'it of wine, or at least a cheekful.}
