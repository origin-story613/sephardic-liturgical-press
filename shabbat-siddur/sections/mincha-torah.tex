\sectiontitle{הוֹצָאַת סֵפֶר תּוֹרָה}

\rubric{Before the ark is opened, say twice:}
\textbf{וַאֲנִי תְפִלָּתִי} לְךָ \Shem{} עֵת רָצוֹן, אֱלֹהִים בְּרב חַסְדֶּךָ, עֲנֵנִי בֶּאֱמֶת יִשְׁעֶךָ:

\textbf{וַאֲנִי תְפִלָּתִי} לְךָ \Shem{} עֵת רָצוֹן, אֱלֹהִים בְּרב חַסְדֶּךָ, עֲנֵנִי בֶּאֱמֶת יִשְׁעֶךָ:

\rubric{Open the ark and say the following. At the words \foreignlanguage{hebrew}{דְסָגִידְנָא קַמֵּהּ} bow slightly toward the Torah. Some add the words in brackets.}
\textbf{בְּרִיךְ שְׁמֵהּ} דְּמָארֵי עָלְמָא, בְּרִיךְ כִּתְרָךְ וְאַתְרָךְ. יְהֵא רְעוּתָךְ, עִם עַמָּךְ יִשְׂרָאֵל לְעָלַם, וּפוּרְקַן יְמִינָךְ אַחְזֵי לְעַמָּךְ בְּבֵית מִקְדָּשָׁךְ, לְאַמְטוּיֵי לָנָא מִטּוּב נְהוֹרָךְ, וּלְקַבְּלָא צְלוֹתָנָא בְּרַחֲמֵי. יְהֵא רַעֲוָא קֳדָמָךְ דְּתוֹרִיךְ לָן חַיִּין בּטִיבוּ, וְלֶהֱוֵי אֲנָא עַבְדָּךְ פְּקִידָא בְּגוֹ צַדִּיקַיָּא, לְמִרְחַם עָלַי וּלְמִנְטַר יָתִי, וְיַת כּל דִּי לִי, וְדִי לְעַמָּךְ יִשְׂרָאֵל. אַנְתְּ הוּא זָן לכֹלָא וּמְפַרְנֵס לְכֹלָא. אַנְתְּ הוּא שַׁלִּיט עַל כֹּלָא, אַנְתְּ הוּא דְשַׁלִּיט עַל מַלְכַיָּא, וּמַלְכוּתָא דִילָךְ הִיא. אֲנָא עַבְדָּא דְקֻדְשָׁא בְּרִיךְ הוּא, דְסָגִידְנָא קַמֵּהּ וּמִן קַמֵּי דִּיקַר אוֹרַיְתֵהּ בְּכָל עִידָּן וְעִידָּן. לָא עַל אֱנָשׁ רָחִיצְנָא, וְלָא עַל בַּר אֱלָהִין סָמִיכְנָא, אֶלָּא בֵאלָהָא דִשְׁמַיָּא, דְּהוּא אֱלָהָא דִקְשׁוֹט, וְאוֹרַיְתֵהּ קְשׁוֹט, וּנְבִיאוֹהִי קְשׁוֹט, וּמַסְגֵּי לְמֶעְבַּד טַבְוָן וּקְשׁוֹט, בֵּהּ אֲנָא רְחִיץ וְלִשְׁמֵהּ יַקִּירָא קַדִּישָׁא אֲנָא אֵמַר תֻּשְׁבְּחָן. יְהֵא רַעֲוָא קֳדָמָךְ דְּתִפְתַּח לִבָּאִי בְּאוֹרַיְתָךְ, [וְתִיהַב לִי בְּנִין דִּכְרִין דְּעַבְדִּין רְעוּתָךְ], וְתַשְׁלִים מִשְׁאֲלִין דְּלִבָּאי וְלִבָּא דְכָל עַמָּךְ יִשְׂרָאֵל לְטַב וּלְחַיִּין ולִשְׁלָם אָמֵן:

\rubric{Take out the Torah scroll and say:}
\textbf{בָּרוּךְ הַמָּקוֹם} שֶׁנָּתַן תּוֹרָה לעַמּוֹ יִשְׂרָאֵל, בָּרוּךְ הוּא: אַשְׁרֵי הָעָם שֶׁכָּכָה לּוֹ, אַשְׁרֵי הָעָם שֶׁ\Shem{} אֱלֹהָיו:

\rubric{As the Torah is carried, the chazzan and congregation say:}
\textbf{גַּדּלוּ} \Shem{לַ} אִתִּי, וּנְרוֹממָה שְׁמוֹ יַחְדָּו: רוֹמְמוּ \Shem{} אֱלֹהֵינוּ והִשְׁתַּחֲווּ לַהֲדֹם רַגְלָיו, קָדוֹשׁ הוּא: רוֹמְמוּ \Shem{} אֱלֹהֵינוּ והִשְׁתַּחֲווּ לְהַר קָדְשׁוֹ, כִּי קָדוֹשׁ \Shem{} אֱלֹהֵינוּ: \Shem{} עֹז לעַמּוֹ יִתֵּן, \Shem{} יְבָרֵךְ אֶת עַמּוֹ בַשָּׁלוֹם:

\textbf{אֵין קָדוֹשׁ} \Shem{כַּ} כִּי אֵין בִּלְתֶּךָ, וְאֵין צוּר כֵּאלֹהֵינוּ: כִּי מִי אֱלוֹהַּ מִבַּלְעֲדֵי \Shem{}, וּמִי צוּר זוּלָתִי אֱלֹהֵינוּ: תּוֹרָה צִוָּה לָנוּ מֹשֶׁה, מוֹרָשָׁה קְהִלַּת יַעֲקֹב: עֵץ חַיִּים הִיא לַמַּחֲזִיקִים בָּהּ, וְתֹמכֶיהָ מְאֻשָּׁר: דְּרָכֶיהָ דַרְכֵי נֹעַם, וְכל נְתִיבֹתֶיהָ שָׁלוֹם: שָׁלוֹם רָב לְאֹהֲבֵי תוֹרָתֶךָ, וְאֵין לָמוֹ מִכְשׁוֹל: \Shem{} עֹז לְעַמּוֹ יִתֵּן, \Shem{} יְבָרֵךְ אֶת עַמּוֹ בַשָּׁלוֹם: כִּי שֵׁם \Shem{} אֶקְרָא, הָבו גֹדֶל לֵאלֹהֵינוּ: הַכֹּל תְּנוּ עֹז לֵאלֹהִים וּתְנוּ כָבוֹד לַתּוֹרָה: יִשְׂמְחוּ הַשָּׁמַיִם וְתָגֵל הָאָרֶץ, וְיֹאמרוּ בַגּוֹיִם \Shem{} מָלָךְ: יִשְׂמַח הַר צִיּוֹן תָּגֵלְנָה בְּנוֹת יְהוּדָה, לְמַעַן מִשְׁפָּטֶיךָ:

\rubric{The Torah is raised and its writing shown to the congregation. All bow and say:}
\textbf{וְזֹאת הַתּוֹרָה}, אֲשֶׁר שָׂם מֹשֶׁה לִפְנֵי בְּנֵי יִשְׂרָאֵל: תּוֹרָה צִוָּה לָנוּ מֹשֶׁה, מוֹרָשָׁה קְהִלַּת יַעֲקֹב: הָאֵל תָּמִים דַּרְכּוֹ, אִמְרַת־\Shem{} צְרוּפָה, מָגֵן הוּא לְכָל הַחוֹסִים בּוֹ:

\rubric{At least ten verses of next week's parashah are read. Half Kaddish is not said after the reading. Then say:}
\textbf{מִזְמוֹר שִׁיר} לְיוֹם הַשַּׁבָּת: טוֹב לְהֹדוֹת \Shem{לַ} וּלְזַמֵּר לְשִׁמְךָ עֶלְיוֹן: לְהַגִּיד בַּבֹּקֶר חַסְדֶּךָ וֶאֱמוּנָתְךָ בַּלֵּילוֹת: עֲלֵי עָשׂוֹר וַעֲלֵי נָבֶל עֲלֵי הִגָּיוֹן בְּכִנּוֹר: כִּי שִׂמַּחְתַּנִי \Shem{} בְּפָעֳלֶךָ בְּמַעֲשֵׂי יָדֶיךָ אֲרַנֵּן: מַה גָּדְלוּ מַעֲשֶׂיךָ \Shem{} מְאֹד עָמְקוּ מַחְשְׁבֹתֶיךָ: אִישׁ בַּעַר לֹא יֵדָע וּכְסִיל לֹא יָבִין אֶת זֹאת: בִּפְרֹחַ רְשָׁעִים כְּמוֹ עֵשֶׂב וַיָּצִיצוּ כָּל פֹּעֲלֵי אָוֶן לְהִשָּׁמְדָם עֲדֵי עַד: וְאַתָּה מָרוֹם לְעֹלָם \Shem{}: כִּי הִנֵּה אֹיְבֶיךָ \Shem{} כִּי הִנֵּה אֹיְבֶיךָ יֹאבֵדוּ יִתְפָּרְדוּ כָּל פֹּעֲלֵי אָוֶן: וַתָּרֶם כִּרְאֵים קַרְנִי בַּלֹּתִי בְּשֶׁמֶן רַעֲנָן: וַתַּבֵּט עֵינִי בְּשׁוּרָי בַּקָּמִים עָלַי מְרֵעִים תִּשְׁמַעְנָה אָזְנָי: צַדִּיק כַּתָּמָר יִפְרָח כְּאֶרֶז בַּלְּבָנוֹן יִשְׂגֶּה: שְׁתוּלִים בְּבֵית \Shem{} בְּחַצְרוֹת אֱלֹהֵינוּ יַפְרִיחוּ: עוֹד יְנוּבוּן בְּשֵׂיבָה דְּשֵׁנִים וְרַעֲנַנִּים יִהְיוּ: לְהַגִּיד כִּי יָשָׁר \Shem{} צוּרִי וְלֹא עַוְלָתָה בּוֹ:

\rubric{Return the Torah to the ark and say:}
\textbf{יְהַללוּ} אֶת שֵׁם \Shem{} כִּי נִשְׂגָּב שְׁמוֹ לְבַדּוֹ, הוֹדוֹ עַל אֶרֶץ וְשָׁמָיִם: וַיָּרֶם קֶרֶן לְעַמּוֹ תְּהִלָּה לְכָל חֲסִידָיו לִבְנֵי יִשְׂרָאֵל עַם קְרֹבוֹ הַלְלוּיָהּ: \Shem{} הוּא הָאֱלֹהִים, \Shem{} הוּא הָאֱלֹהִים, בַּשָּׁמַיִם מִמַּעַל, וְעַל הָאָרֶץ מִתָּחַת, אֵין עוֹד: אֵין כָּמוֹךָ בָאֱלֹהִים אֲדֹנָי וְאֵין כְּמַעֲשֶׂיךָ: הֲשִׁיבֵנוּ \Shem{} אֵלֶיךָ ונָשׁוּבָה, חַדֵּשׁ יָמֵינוּ כְּקֶדֶם:

\textbf{כִּי לֶקַח טוֹב} נָתַתִּי לָכֶם תּוֹרָתִי אַל־תַּעֲזוֹבוּ: תִּכּוֹן תְּפִלָּתִי קטֹרֶת לְפָנֶיךָ, מַשְׂאַת כַּפַּי מִנְחַת עָרֶב:

\rubric{The chazzan says Half Kaddish:}
