\sectiontitle{עֲמִידָה}\label{mincha-amidah}

\textbf{אֲדֹנָי}, שְׂפָתַי תִּפְתָּח וּפִי יַגִּיד תְּהִלָּתֶךָ:

\textbf{בָּרוּךְ} אַתָּה \Shem{}, אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ, אֱלֹהֵי אַבְרָהָם, אֱלֹהֵי יִצְחָק וֵאלֹהֵי יַעֲקֹב. הָאֵל הַגָּדוֹל, הַגִּבּוֹר וְהַנּוֹרָא, אֵל עֶלְיוֹן, גּוֹמֵל חֲסָדִים טוֹבִים, קוֹנֵה הַכֹּל, וְזוֹכֵר חַסְדֵי אָבוֹת, וּמֵבִיא גוֹאֵל לִבְנֵי בְנֵיהֶם לְמַעַן שְׁמוֹ, בְּאַהֲבָה,

\begin{seasonal}{On Shabbat Shuvah add:}
זָכְרֵנוּ לְחַיִּים, מֶלֶךְ חָפֵץ בַּחַיִּים, כָּתְבֵנוּ בְּסֵפֶר חַיִּים, לְמַעַנְךָ אֱלֹהִים חַיִּים.
\end{seasonal}

מֶלֶךְ עוֹזֵר וּמוֹשִׁיעַ וּמָגֵן, בָּרוּךְ אַתָּה \Shem{}, מָגֵן אַבְרָהָם:

\textbf{אַתָּה גִּבּוֹר} לְעוֹלָם, אֲדֹנָי. מְחַיֵּה מֵתִים אַתָּה, רַב לְהוֹשִׁיעַ.

\begin{seasonal}{In summer:}
מוֹרִיד הַטָּל.
\end{seasonal}

\begin{seasonal}{In winter:}
מַשִּׁיב הָרוּחַ וּמוֹרִיד הַגֶּשֶׁם.
\end{seasonal}

מְכַלְכֵּל חַיִּים בְּחֶסֶד, מְחַיֵּה מֵתִים בְּרַחֲמִים רַבִּים, סוֹמֵךְ נוֹפְלִים, וְרוֹפֵא חוֹלִים, וּמַתִּיר אֲסוּרִים, וּמְקַיֵּם אֱמוּנָתוֹ לִישֵׁנֵי עָפָר. מִי כָמוֹךָ בַעַל גְּבוּרוֹת, וּמִי דוֹמֶה לָּךְ, מֶלֶךְ מֵמִית וּמְחַיֶּה וּמַצְמִיחַ יְשׁוּעָה.

\begin{seasonal}{On Shabbat Shuvah add:}
מִי כָמוֹךָ אָב הָרַחֲמָן, זוֹכֵר יְצוּרָיו בְּרַחֲמִים לְחַיִּים
\end{seasonal}

וְנֶאֱמָן אַתָּה לְהַחֲיוֹת מֵתִים. בָּרוּךְ אַתָּה \Shem{}, מְחַיֵּה הַמֵּתִים:

\rubric{In the chazzan's repetition, Kedushah is said here. The congregation rises on the toes at each \foreignlanguage{hebrew}{קָדוֹשׁ} and also at \foreignlanguage{hebrew}{בָּרוּךְ} and \foreignlanguage{hebrew}{יִמְלֹךְ} in the verses below.}
נַקְדִּישָׁךְ וְנַעֲרִיצָךְ כְּנֹעַם שִׂיחַ סוֹד שַׂרְפֵי קֹדֶשׁ, הַמְשַׁלְּשִׁים לְךָ קְדֻשָּׁה. כַּכָּתוּב עַל יַד נְבִיאֶךָ: וְקָרָא זֶה אֶל זָה וְאָמַר: \textbf{קָדוֹש, קָדוֹש, קָדוֹש, \Shem{} צְבָאוֹת. מְלֹא כָל הָאָרֶץ כְּבוֹדוֹ.} לְעֻמָּתָם מְשַׁבְּחִים וְאוֹמְרִים: \textbf{בָּרוּךְ כְּבוֹד \Shem{} מִמְּקוֹמוֹ.} וּבְדִבְרֵי קָדְשְׁךָ כָּתוּב לֵאמֹר: \textbf{יִמְלֹךְ \Shem{} לְעוֹלָם אֱלֹהַיִךְ צִיּוֹן לְדֹר וָדֹר, הַלְלוּיָהּ.}

\textbf{אַתָּה קָדוֹשׁ}, וְשִׁמְךָ קָדוֹשׁ, וּקְדוֹשִׁים בְּכָל יוֹם יְהַלְלוּךָ סֶּלָה. בָּרוּךְ אַתָּה \Shem{}, הָאֵל הַקָּדוֹשׁ:

\begin{seasonal}{On Shabbat Shuvah, end instead:}
הַמֶּלֶךְ הַקָּדוֹשׁ:
\end{seasonal}



\rubric{Some add the word in brackets.}
\textbf{אַתָּה אֶחָד} וְשִׁמְךָ אֶחָד, וּמִי כְעַמְּךָ כְּיִשְׂרָאֵל גוֹי אֶחָד בָּאָרֶץ. תִּפְאֶרֶת גְּדֻלָּה, וַעֲטֶרֶת יְשׁוּעָה, יוֹם מְנוּחָה וּקְדֻשָּׁה לְעַמְּךָ נָתַתָּ. אַבְרָהָם יָגֵל, יִצְחָק יְרַנֵּן, יַעֲקֹב וּבָנָיו יָנוּחוּ בוֹ. מְנוּחַת אַהֲבָה וּנְדָבָה, מְנוּחַת אֱמֶת וֶאֱמוּנָה, מְנוּחַת שָׁלוֹם הַשְׁקֵט וָבֶטַח, מְנוּחָה שְׁלֵמָה שֶׁאַתָּה [הוּא] רוֹצֶה בָהּ. יַכִּירוּ בָנֶיךָ וְיֵדעוּ כִּי מֵאִתּךָ הִיא מְנוּחָתָם, ועַל מְנוּחָתָם יַקְדִּישׁוּ אֶת שְׁמֶךָ:

\textbf{אֱלֹהֵינוּ} וֵאלֹהֵי אֲבוֹתֵינוּ, רְצֵה נָא בִּמְנוּחָתֵנוּ, קַדְּשֵׁנוּ בְּמִצְוֹתֶיךָ, שִׂים חֶלְקֵנוּ בְּתוֹרָתָךְ, שַׂבְּעֵנוּ מִטּוּבָךְ, שַׂמֵּחַ נַפְשֵׁנוּ בִּישׁוּעָתָךְ, וְטַהֵר לִבֵּנוּ לעָבְדְּךָ בֶּאֱמֶת, וְהַנְחִילֵנוּ \Shem{} אֱלֹהֵינוּ בְּאַהֲבָה וּבְרָצוֹן שַׁבָּת קָדְשֶׁךָ, וְיָנוּחוּ בוֹ כָּל יִשְׂרָאֵל מְקַדְּשֵׁי שְׁמֶךָ. בָּרוּךְ אַתָּה \Shem{}, מְקַדֵּשׁ הַשַּׁבָּת:

\textbf{רְצֵה} \Shem{} אֱלֹהֵינוּ בְּעַמְּךָ יִשְׂרָאֵל, וּלִתְפִלָּתָם שְׁעֵה, וְהָשֵׁב הָעֲבוֹדָה לִדְבִיר בֵּיתֶךָ, וְאִשֵּׁי יִשְׂרָאֵל וּתְפִלָּתָם מְהֵרָה בְאַהֲבָה תְּקַבֵּל בְּרָצוֹן, וּתְהִי לְרָצוֹן תָּמִיד עֲבוֹדַת יִשְׂרָאֵל עַמֶּךָ:

\begin{seasonal}{On Rosh Hodesh and Hol HaMoed:}
אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ, יַעֲלֶה וְיָבֹא, יַגִּיעַ יֵרָאֶה וְיֵרָצֶה, יִשָּׁמַע יִפָּקֵד וְיִזָּכֵר, זִכְרוֹנֵנוּ וְזִכְרוֹן אֲבוֹתֵינוּ, זִכְרוֹן יְרוּשָׁלַיִם עִירָךְ, וְזִכְרוֹן מָשִׁיחַ בֶּן דָּוִד עַבְדָּךְ, וְזִכְרוֹן כָּל עַמְּךָ בֵּית יִשְׂרָאֵל לְפָנֶיךָ, לִפְלֵטָה, לְטוֹבָה, לְחֵן, לְחֶסֶד וּלְרַחֲמִים לְחַיִּים טוֹבִים וּלְשָׁלוֹם בְּיוֹם:

\rubric{On Rosh Hodesh:}
רֹאשׁ חֹדֶשׁ הַזֶּה,

\rubric{On Hol HaMoed Pesah:}
חַג הַמַּצּוֹת הַזֶּה, בְּיוֹם מִקְרָא קֹדֶשׁ הַזֶּה,

\rubric{On Hol HaMoed Sukkot:}
חַג הַסֻּכּוֹת הַזֶּה, בְּיוֹם מִקְרָא קֹדֶשׁ הַזֶּה,

\rubric{Continue:}
לְרַחֵם בּוֹ עָלֵינוּ וּלְהוֹשִׁיעֵנוּ. זָכְרֵנוּ \Shem{} אֱלֹהֵינוּ בּוֹ לְטוֹבָה, וּפָּקְדֵנוּ בוֹ לִבְרָכָה, וְהוֹשִׁיעֵנוּ בוֹ לְחַיִּים טוֹבִים, בִּדְבַר יְשׁוּעָה וְרַחֲמִים. חוּס וְחָנֵּנוּ, וַחֲמֹל וְרַחֵם עָלֵינוּ, וְהוֹשִׁיעֵנוּ כִּי אֵלֶיךָ עֵינֵינוּ, כִּי אֵל מֶלֶךְ חַנּוּן וְרַחוּם אָתָּה:
\end{seasonal}

וְאַתָּה, בְּרַחֲמֶיךָ הָרַבִּים, תַּחְפֹּץ בָּנוּ וְתִרְצֵנוּ. וְתֶחֱזֶינָה עֵינֵינוּ בְּשׁוּבְךָ לְצִיּוֹן בְּרַחֲמִים. בָּרוּךְ אַתָּה \Shem{}, הַמַּחֲזִיר שְׁכִינָתוֹ לְצִיּוֹן:

\rubric{Bow at \foreignlanguage{hebrew}{מוֹדִים} as you begin.}
\textbf{מוֹדִים} אֲנַחְנוּ לָךְ, שָׁאַתָּה הוּא \Shem{} אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ לְעוֹלָם וָעֶד, צוּרֵנוּ צוּר חַיֵּינוּ וּמָגֵן יִשְׁעֵנוּ אַתָּה הוּא לְדֹר וָדֹר. נוֹדֶה לְּךָ וּנְסַפֵּר תְּהִלָּתֶךָ, עַל חַיֵּינוּ הַמְּסוּרִים בְּיָדֶיךָ, וְעַל נִשְׁמוֹתֵינוּ הַפְּקוּדוֹת לָךְ, וְעַל נִסֶּיךָ שֶׁבְּכָל יוֹם עִמָּנוּ, וְעַל נִפְלְאוֹתֶיךָ וְטוֹבוֹתֶיךָ שֶׁבְּכָל עֵת, עֶרֶב וָבֹקֶר וְצָהֳרָיִם. הַטּוֹב, כִּי לֹא כָלוּ רַחֲמֶיךָ, הַמְּרַחֵם, כִּי לֹא תַמּוּ חֲסָדֶיךָ, כִּי מֵעוֹלָם קִוִּינוּ לָךְ:

\rubric{In the chazzan's repetition, as he begins \foreignlanguage{hebrew}{מוֹדִים} the congregation says:}
\textbf{מוֹדִים} אֲנַחְנוּ לָךְ, שָׁאַתָּה הוּא \Shem{} אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ, אֱלֹהֵי כָל בָּשָׁר, יוֹצְרֵנוּ יוֹצֵר בְּרֵאשִׁית. בְּרָכוֹת וְהוֹדָאוֹת לְשִׁמְךָ הַגָּדוֹל וְהַקָּדוֹשׁ, עַל שֶׁהֶחֱיִיתָנוּ וְקִיַּמְתָּנוּ. כֵּן תְּחַיֵּנוּ וּתְחָנֵּנוּ וְתֶאֱסוֹף גָּלֻיּוֹתֵינוּ לְחַצְרוֹת קָדְשֶׁךָ, לִשְׁמֹר חֻקֶּיךָ וְלַעֲשׂוֹת רְצוֹנְךָ וּלְעָבְדְךָ בְלֵבָב שָׁלֵם, עַל שֶׁאֲנַחְנוּ מוֹדִים לָךְ, בָּרוּךְ אֵל הַהוֹדָאוֹת.

\begin{seasonal}{On Hanukkah and Purim:}
\textbf{עַל הַנִּסִּים} וְעַל הַפֻּרְקָן וְעַל הַגְּבוּרוֹת וְעַל הַתְּשׁוּעוֹת וְעַל הַנִּפְלָאוֹת וְעַל הַנֶּחָמוֹת שֶׁעָשִׂיתָ לַאֲבוֹתֵינוּ בַּיָּמִים הָהֵם בַּזְּמַן הַזֶּה.

\rubric{On Hanukkah:}
\textbf{בִּימֵי מַתִּתְיָה} בֶן יוֹחָנָן כֹּהֵן גָּדוֹל חַשְׁמוֹנָאִי וּבָנָיו כְּשֶׁעָמְדָה מַלְכוּת יָוָן הָרְשָׁעָה עַל עַמְּךָ יִשְׂרָאֵל לְשַׁכְּחָם תּוֹרָתָךְ וּלְהַעֲבִירָם מֵחֻקֵּי רְצוֹנָךְ וְאַתָּה בְּרַחֲמֶיךָ הָרַבִּים עָמַדְתָּ לָהֶם בְּעֵת צָרָתָם רַבְתָּ אֶת רִיבָם דַּנְתָּ אֶת דִּינָם נָקַמְתָּ אֶת נִקְמָתָם מָסַרְתָּ גִבּוֹרִים בְּיַד חַלָּשִׁים וְרַבִּים בְּיַד מְעַטִּים וּטְמֵאִים בְּיַד טְהוֹרִים וּרְשָׁעִים בְּיַד צַדִּיקִים וְזֵדִים בְּיַד עוֹסְקֵי תוֹרָתֶךָ לְךָ עָשִׂיתָ שֵׁם גָּדוֹל וְקָדוֹשׁ בְּעוֹלָמָךְ וּלְעַמְּךָ יִשְׂרָאֵל עָשִׂיתָ תְּשׁוּעָה גְדוֹלָה וּפֻרְקָן כְּהַיּוֹם הַזֶּה וְאַחַר כָּךְ בָּאוּ בָנֶיךָ לִדְבִיר בֵּיתֶךָ וּפִנּוּ אֶת הֵיכָלֶךָ וְטִהֲרוּ אֶת מִקְדָּשֶׁךָ וְהִדְלִיקוּ נֵרוֹת בְּחַצְרוֹת קָדְשֶׁךָ וְקָבְעוּ שְׁמוֹנַת יְמֵי חֲנֻכָּה אֵלּוּ בְּהַלֵּל גָּמוּר וּבְהוֹדָאָה, וְעָשִׂיתָ עִמָּהֶם נִסִּים וְנִפְלָאוֹת וְנוֹדֶה לְשִׁמְךָ הַגָּדוֹל סֶלָה:

\rubric{On Purim:}
\textbf{בִּימֵי מָרְדְּכַי} וְאֶסְתֵּר בְּשׁוּשַׁן הַבִּירָה כְּשֶׁעָמַד עֲלֵיהֶם הָמָן הָרָשָׁע בִּקֵּשׁ לְהַשְׁמִיד לַהֲרוֹג וּלְאַבֵּד אֶת כָּל הַיְּהוּדִים מִנַּעַר וְעַד זָקֵן טַף וְנָשִׁים בְּיוֹם אֶחָד בִּשְׁלֹשָׁה עָשָׂר לְחֹדֶשׁ שְׁנֵים עָשָׂר הוּא חֹדֶשׁ אֲדָר וּשְׁלָלָם לָבוֹז וְאַתָּה בְּרַחֲמֶיךָ הָרַבִּים הֵפַרְתָּ אֶת עֲצָתוֹ וְקִלְקַלְתָּ אֶת מַחֲשַׁבְתּוֹ וַהֲשֵׁבוֹתָ לוֹ גְמוּלוֹ בְרֹאשׁוֹ וְתָלוּ אוֹתוֹ וְאֶת בָּנָיו עַל הָעֵץ וְעָשִׂיתָ עִמָּהֶם נִסִּים וְנִפְלָאוֹת וְנוֹדֶה לְשִׁמְךָ הַגָּדוֹל סֶלָה:
\end{seasonal}

\rubric{Bow at the closing \foreignlanguage{hebrew}{בָּרוּךְ} and straighten up at the Name.}
וְעַל כֻּלָּם יִתְבָּרַךְ וְיִתְרוֹמַם וְיִתְנַשֵּׂא תָמִיד, שִׁמְךָ מַלְכֵּנוּ, לְעוֹלָם וָעֶד. וְכָל הַחַיִּים יוֹדוּךָ סֶּלָה:

\begin{seasonal}{On Shabbat Shuvah add:}
וּכְתֹב לְחַיִּים טוֹבִים כָּל בְּנֵי בְרִיתֶךָ.
\end{seasonal}

וִיהַלְלוּ וִיבָרְכוּ אֶת שִׁמְךָ הַגָּדוֹל, בֶּאֱמֶת, לְעוֹלָם כִּי טוֹב, הָאֵל יְשׁוּעָתֵנוּ וְעֶזְרָתֵנוּ סֶלָה, הָאֵל הַטּוֹב. בָּרוּךְ אַתָּה \Shem{}, הַטּוֹב שִׁמְךָ וּלְךָ נָאֶה לְהוֹדוֹת:

\textbf{שִׂים שָׁלוֹם} טוֹבָה וּבְרָכָה, חַיִּים, חֵן וָחֶסֶד, צְדָקָה וְרַחֲמִים, עָלֵינוּ וְעַל כָּל יִשְׂרָאֵל עַמֶּךָ. וּבָרְכֵנוּ אָבִינוּ, כֻּלָּנוּ כְאֶחָד בְּאוֹר פָּנֶיךָ, כִּי בְאוֹר פָּנֶיךָ נָתַתָּ לָּנוּ \Shem{} אֱלֹהֵינוּ, תּוֹרָה וְחַיִּים, אַהֲבָה וָחֶסֶד, צְדָקָה וְרַחֲמִים, בְּרָכָה וְשָׁלוֹם. וְטוֹב בְּעֵינֶיךָ לְבָרְכֵנוּ וּלְבָרֵךְ אֶת כָּל עַמְּךָ יִשְׂרָאֵל בְּרֹב עֹז וְשָׁלוֹם.

\begin{seasonal}{On Shabbat Shuvah add:}
וּבְסֵפֶר חַיִּים, בְּרָכָה וְשָׁלוֹם, וּפַרְנָסָה טוֹבָה וִישׁוּעָה וְנֶחָמָה, וּגְזֵרוֹת טוֹבוֹת, נִזָּכֵר וְנִכָּתֵב לְפָנֶיךָ, אֲנַחְנוּ וְכָל עַמְּךָ בֵּית יִשְׂרָאֵל, לְחַיִּים טוֹבִים וּלְשָׁלוֹם.
\end{seasonal}

בָּרוּךְ אַתָּה \Shem{}, הַמְּבָרֵךְ אֶת עַמּוֹ יִשְׂרָאֵל בַּשָּׁלוֹם. אָמֵן:

\textbf{יִהְיוּ לְרָצוֹן} אִמְרֵי פִי וְהֶגְיוֹן לִבִּי לְפָנֶיךָ \Shem{} צוּרִי וְגֹאֲלִי:

\textbf{אֱלֹהַי}, נְצֹר לְשׁוֹנִי מֵרָע, וּשְׂפָתַי מִדַּבֵּר מִרְמָה, וְלִמְקַלְלַי נַפְשִׁי תִדֹּם, וְנַפְשִׁי כֶּעָפָר לַכֹּל תִּהְיֶה, פְּתַח לִבִּי בְּתוֹרָתֶךָ, וְאַחֲרֵי מִצְוֹתֶיךָ תִּרְדֹּף נַפְשִׁי. וְכָל הַקָּמִים עָלַי לְרָעָה, מְהֵרָה הָפֵר עֲצָתָם וְקַלְקֵל מַחְשְבוֹתָם. יִהְיוּ כְּמֹץ לִפְנֵי רוּחַ. וּמַלְאַךְ \Shem{} דּוֹחֶה. עֲשֵׂה לְמַעַן שְׁמָךְ, עֲשֵׂה לְמַעַן יְמִינָךְ, עֲשֵׂה לְמַעַן תּוֹרָתָךְ, עֲשֵׂה לְמַעַן קְדֻשָּׁתָךְ, לְמַעַן יֵחָלְצוּן יְדִידֶיךָ. הוֹשִׁיעָה יְמִינְךָ וַעֲנֵנִי.

\rubric{It is customary to say here a verse that begins and ends with the first and last letters of one's own name.}
\textbf{יִהְיוּ לְרָצוֹן} אִמְרֵי פִי וְהֶגְיוֹן לִבִּי לְפָנֶיךָ \Shem{} צוּרִי וְגֹאֲלִי:

\rubric{On Shabbat Shuvah, \foreignlanguage{hebrew}{עֹשֶׂה הַשָּׁלוֹם} is said instead.}
\textbf{עֹשֶׂה שָׁלוֹם} בִּמְרוֹמָיו, הוּא בְרַחֲמָיו יַעֲשֶׂה שָׁלוֹם עָלֵינוּ, וְעַל כָּל עַמּוֹ יִשְׂרָאֵל, וְאִמְרוּ אָמֵן:

\textbf{יְהִי רָצוֹן} מִלְּפָנֶיךָ \Shem{} אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ, שֶׁתִּבְנֶה בֵּית הַמִּקְדָּשׁ בִּמְהֵרָה בְּיָמֵינוּ, וְתֵן חֶלְקֵנוּ בְּתוֹרָתָךְ, לַעֲשׂוֹת חֻקֵּי רְצוֹנָךְ וּלְעָבְדָךְ בְּלֵבָב שָׁלֵם:
