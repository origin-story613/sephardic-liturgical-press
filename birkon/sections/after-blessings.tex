\sectiontitle{בְּרָכָה מֵעֵין שָׁלֹשׁ}

\rubric{After grain foods other than bread, wine or grape juice, or the fruits
of the Land of Israel (grapes, figs, pomegranates, olives, dates):}
\textbf{בָּרוּךְ} אַתָּה \Shem{} אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם,

\begin{seasonal}{After grain foods:}
עַל הַמִּחְיָה וְעַל הַכַּלְכָּלָה
\end{seasonal}

\begin{seasonal}{After wine or grape juice:}
עַל הַגֶּפֶן וְעַל פְּרִי הַגֶּפֶן
\end{seasonal}

\begin{seasonal}{After the fruits of the Land of Israel:}
עַל הָעֵץ וְעַל פְּרִי הָעֵץ
\end{seasonal}

וְעַל תְּנוּבַת הַשָּׂדֶה, וְעַל אֶרֶץ חֶמְדָּה טוֹבָה וּרְחָבָה, שֶׁרָצִיתָ וְהִנְחַלְתָּ
לַאֲבוֹתֵינוּ, לֶאֱכוֹל מִפִּרְיָהּ וְלִשְׂבּוֹעַ מִטּוּבָהּ. רַחֵם \Shem{} אֱלֹהֵינוּ עָלֵינוּ
וְעַל יִשְׂרָאֵל עַמֶּךָ, וְעַל יְרוּשָׁלַיִם עִירֶךָ, וְעַל הַר צִיּוֹן מִשְׁכַּן כְּבוֹדֶךָ, וְעַל
מִזְבְּחֶךָ וְעַל הֵיכָלֶךָ. וּבְנֵה יְרוּשָׁלַיִם עִיר הַקֹּדֶשׁ בִּמְהֵרָה בְיָמֵינוּ, וְהַעֲלֵנוּ
לְתוֹכָהּ, וְשַׂמְּחֵנוּ בְּבִנְיָנָהּ, וּנְבָרֶכְךָ עָלֶיהָ בִּקְדֻשָּׁה וּבְטָהֳרָה,

\begin{seasonal}{On Shabbat. Some say \foreignlanguage{hebrew}{וּרְצֵה וְהַחֲלִיצֵנוּ} in place of the first word:}
וְנַחֲמֵנוּ בְּיוֹם הַשַּׁבָּת הַזֶּה,
\end{seasonal}

\begin{seasonal}{On Rosh Hodesh:}
וְזָכְרֵנוּ לְטוֹבָה בְּיוֹם רֹאשׁ הַחֹדֶשׁ הַזֶּה,
\end{seasonal}

\begin{seasonal}{On Pesah:}
וְשַׂמְּחֵנוּ בְּיוֹם חַג הַמַּצּוֹת הַזֶּה, בְּיוֹם מִקְרָא קֹדֶשׁ הַזֶּה,
\end{seasonal}

\begin{seasonal}{On Shavuot:}
וְשַׂמְּחֵנוּ בְּיוֹם חַג הַשָּׁבוּעוֹת הַזֶּה, בְּיוֹם מִקְרָא קֹדֶשׁ הַזֶּה,
\end{seasonal}

\begin{seasonal}{On Sukkot:}
וְשַׂמְּחֵנוּ בְּיוֹם חַג הַסֻּכּוֹת הַזֶּה, בְּיוֹם מִקְרָא קֹדֶשׁ הַזֶּה,
\end{seasonal}

\begin{seasonal}{On Shemini Atzeret and Simhat Torah:}
וְשַׂמְּחֵנוּ בְּיוֹם שְׁמִינִי חַג עֲצֶרֶת הַזֶּה, בְּיוֹם מִקְרָא קֹדֶשׁ הַזֶּה,
\end{seasonal}

כִּי אַתָּה טוֹב וּמֵטִיב לַכֹּל, וְנוֹדֶה לְּךָ עַל הָאָרֶץ

\begin{seasonal}{After grain foods:}
וְעַל הַמִּחְיָה וְעַל הַכַּלְכָּלָה.
\end{seasonal}

\begin{seasonal}{After wine or grape juice:}
וְעַל הַגֶּפֶן וְעַל פְּרִי הַגֶּפֶן.
\end{seasonal}

\begin{seasonal}{After the fruits of the Land of Israel:}
וְעַל הָעֵץ וְעַל פְּרִי הָעֵץ.
\end{seasonal}

בָּרוּךְ אַתָּה \Shem{}, עַל הָאָרֶץ

\begin{seasonal}{After grain foods:}
וְעַל הַמִּחְיָה וְעַל הַכַּלְכָּלָה.
\end{seasonal}

\begin{seasonal}{After wine or grape juice:}
וְעַל הַגֶּפֶן וְעַל פְּרִי הַגֶּפֶן.
\end{seasonal}

\begin{seasonal}{After the fruits of the Land of Israel:}
וְעַל הָעֵץ וְעַל פְּרִי הָעֵץ.
\end{seasonal}

\sectiontitle{בּוֹרֵא נְפָשׁוֹת}

\rubric{After all other foods and drinks:}
\textbf{בָּרוּךְ} אַתָּה \Shem{} אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, בּוֹרֵא נְפָשׁוֹת רַבּוֹת וְחֶסְרוֹנָן, עַל
כָּל מַה שֶּׁבָּרָאתָ, לְהַחֲיוֹת בָּהֶם נֶפֶשׁ כָּל חַי. בָּרוּךְ חַי הָעוֹלָמִים.
