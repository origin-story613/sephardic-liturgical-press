\sectiontitle{הַבְדָּלָה}\label{havdalah}

\textbf{רִאשׁוֹן לְצִיּוֹן} הִנֵּה הִנָּם, וְלִירוּשָׁלַיִם מְבַשֵּׂר אֶתֵּן: אַל תִּשְׂמְחִי אוֹיַבְתִּי לִי כִּי נָפַלְתִּי קַמְתִּי, כִּי אֵשֵׁב בַּחוֹשֶׁךְ \Shem{} אוֹר לִי: לַיְּהוּדִים הָיְתָה אוֹרָה וְשִׂמְחָה, וְשָׂשֹׂן וִיקָר: וַיְהִי דָוִד לְכָל דְּרָכָיו מַשְׂכִּיל \Shem{וַ} עִמּוֹ: וְנֹחַ מָצָא חֵן בְּעֵינֵי \Shem{}: כֵּן נִמְצָא חֵן וְשֵׂכֶל טוֹב בְּעֵינֵי אֱלֹהִים וְאָדָם:

\textbf{קוּמִי אוֹרִי} כִּי בָא אוֹרֵךְ, וּכְבוֹד \Shem{} עָלַיִךְ זָרַח: כִּי הִנֵּה הַחֹשֶׁךְ יְכַסֶּה אֶרֶץ וַעֲרָפֶל לְאֻמִּים, וְעָלַיִךְ יִזְרַח \Shem{} וּכְבוֹדוֹ עָלַיִךְ יֵרָאֶה:

\rubric{Take the cup of wine and say:}
\textbf{כּוֹס ישׁוּעוֹת} אֶשָּׂא, וּבְשֵׁם \Shem{} אֶקְרָא:

אָנָּא \Shem{} הוֹשִׁיעָה נָּא:

אָנָּא \Shem{} הוֹשִׁיעָה נָּא:

אָנָּא \Shem{} הַצְלִיחָה נָא:

אָנָּא \Shem{} הַצְלִיחָה נָא:

\textbf{הַצְלִיחֵנוּ}, הַצְלַח דְּרָכֵינוּ, הַצְלַח לִמּוּדֵינוּ, וּשְׁלַח בְּרָכָה רְוָחָה וְהַצְלָחָה בְּכָל מַעֲשֵׂה יָדֵינוּ, כְּדִכְתִיב יִשָּׂא בְרָכָה מֵאֵת \Shem{} וּצְדָקָה מֵאֱלֹהֵי יִשְׁעוֹ: לַיְּהוּדִים הָיְתָה אוֹרָה וְשִׂמְחָה, וְשָׂשֹׂן וִיקָר: וּכְתִיב וַיְהִי דָוִד לְכָל דְּרָכָיו מַשְׂכִּיל, \Shem{וַ} עִמּוֹ: כֵּן יִהְיֶה עִמָּנוּ תָמִיד:

\rubric{The one making Havdalah says:}
סַבְרֵי מָרָנָן.

\rubric{The others respond:}
לְחַיִּים.

\textbf{בָּרוּךְ} אַתָּה \Shem{}, אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, בּוֹרֵא פְּרִי הַגֶּפֶן:

\rubric{On Motzei Shabbat only, bless the spices and the fire. On Motzei Yom Tov that is not Motzei Shabbat, omit them.}

\begin{seasonal}{Over fragrant wood or bark:}
\textbf{בָּרוּךְ} אַתָּה \Shem{}, אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, בּוֹרֵא עֲצֵי בְשָׂמִים:
\end{seasonal}

\begin{seasonal}{Over fragrant herbs:}
\textbf{בָּרוּךְ} אַתָּה \Shem{}, אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, בּוֹרֵא עִשְׂבֵּי בְשָׂמִים:
\end{seasonal}

\begin{seasonal}{Over other spices:}
\textbf{בָּרוּךְ} אַתָּה \Shem{}, אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, בּוֹרֵא מִינֵי בְשָׂמִים:
\end{seasonal}

\rubric{Curl the fingers of the right hand over the thumb, look at the fingernails by the light of the flame, and say:}
\textbf{בָּרוּךְ} אַתָּה \Shem{}, אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, בּוֹרֵא מְאוֹרֵי הָאֵשׁ:

\textbf{בָּרוּךְ} אַתָּה \Shem{}, אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, הַמַּבְדִּיל בֵּין קֹדֶשׁ לְחוֹל וּבֵין אוֹר לְחֹשֶׁךְ וּבֵין יִשְׂרָאֵל לָעַמִּים וּבֵין יוֹם הַשְּׁבִיעִי לְשֵׁשֶׁת יְמֵי הַמַּעֲשֶׂה. בָּרוּךְ אַתָּה \Shem{}, הַמַּבְדִּיל בֵּין קֹדֶשׁ לְחוֹל:

\rubric{Drink a revi'it of wine, then say the blessing after wine (page~\pageref{meein-shalosh}).}
