\sectiontitle{הַבְדָּלָה}\label{havdalah}

\rubric{Lift the cup of wine and say:}
\textbf{רִאשׁוֹן לְצִיּוֹן} הִנֵּה הִנָּם, וְלִירוּשָׁלַיִם מְבַשֵּׂר אֶתֵּן:

\textbf{כּוֹס יְשׁוּעוֹת} אֶשָּׂא, וּבְשֵׁם \Shem{} אֶקְרָא:

\rubric{Each line is said twice.}
אָנָּא \Shem{} הוֹשִׁיעָה נָּא:

אָנָּא \Shem{} הַצְלִיחָה נָא:

\rubric{Some add the last word, in brackets.}
\textbf{הַצְלִיחֵנוּ}, הַצְלִיחַ דְּרָכֵנוּ, הַצְלִיחַ לִמּוּדֵנוּ, וּשְׁלַח בְּרָכָה רְוָחָה וְהַצְלָחָה בְּכָל מַעֲשֵׂה יָדֵינוּ, כְּדִכְתִיב יִשָּׂא בְרָכָה מֵאֵת \Shem{} וּצְדָקָה מֵאֱלֹהֵי יִשְׁעוֹ: לַיְּהוּדִים הָיְתָה אוֹרָה וְשִׂמְחָה, וְשָׂשֹׂן וִיקָר: וּכְתִיב וַיְהִי דָוִד לְכָל דְּרָכָיו מַשְׂכִּיל, \Shem{וַ} עִמּוֹ: כֵּן יִהְיֶה עִמָּנוּ [תָּמִיד]:

\rubric{The one making Havdalah says:}
סַבְרִי מָרָנָן.

\rubric{The others respond (some omit the letter in brackets):}
[לְ]חַיִּים.

\textbf{בָּרוּךְ} אַתָּה \Shem{}, אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, בּוֹרֵא פְּרִי הַגֶּפֶן:

\rubric{On Motzei Shabbat only (not on Motzei Yom Tov), take the cup in the left hand and the spices in the right, and say:}

\begin{seasonal}{Over cloves or other spices:}
\textbf{בָּרוּךְ} אַתָּה \Shem{}, אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, בּוֹרֵא מִינֵי בְשָׂמִים:
\end{seasonal}

\begin{seasonal}{Over fragrant branches, such as myrtle:}
\textbf{בָּרוּךְ} אַתָּה \Shem{}, אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, בּוֹרֵא עֲצֵי בְשָׂמִים:
\end{seasonal}

\begin{seasonal}{Over fragrant herbs:}
\textbf{בָּרוּךְ} אַתָּה \Shem{}, אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, בּוֹרֵא עִשְׂבֵּי בְשָׂמִים:
\end{seasonal}

\begin{seasonal}{Over a lemon:}
\textbf{בָּרוּךְ} אַתָּה \Shem{}, אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, הַנּוֹתֵן רֵיחַ טוֹב בַּפֵּרוֹת:
\end{seasonal}

\rubric{By the light of a candle, say:}
\textbf{בָּרוּךְ} אַתָּה \Shem{}, אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, בּוֹרֵא מְאוֹרֵי הָאֵשׁ:

\textbf{בָּרוּךְ} אַתָּה \Shem{}, אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, הַמַּבְדִּיל בֵּין קֹדֶשׁ לְחוֹל וּבֵין אוֹר לְחוֹשֶׁךְ וּבֵין יִשְׂרָאֵל לָעַמִּים וּבֵין יוֹם הַשְּׁבִיעִי לְשֵׁשֶׁת יְמֵי הַמַּעֲשֶׂה. בָּרוּךְ אַתָּה \Shem{}, הַמַּבְדִּיל בֵּין קֹדֶשׁ לְחוֹל:

\rubric{Drink a revi'it of wine, then say the blessing after wine (page~\pageref{meein-shalosh}).}
